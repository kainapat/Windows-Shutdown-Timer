- [ ] **Step 3: Implement the minimal fail-closed shim and capture helpers**

capture_ui_states.py must begin with this contract:

~~~python
from __future__ import annotations
import os
os.environ.setdefault("QT_QPA_PLATFORM", "windows")

import subprocess as std_subprocess
import sys
import time
from pathlib import Path

APP_ROOT = Path(__file__).resolve().parents[3]
sys.path.insert(0, str(APP_ROOT))

from PySide6.QtCore import QDateTime
from PySide6.QtWidgets import QApplication, QMessageBox
import shutdown_timer

class FakeSubprocessRun:
    CalledProcessError = std_subprocess.CalledProcessError

    def __init__(self) -> None:
        self.calls: list[list[str]] = []
        self.spawned_processes = 0

    def run(self, args, *pos, **kwargs):
        cmd = [str(x) for x in args]
        if not cmd or cmd[0].lower() not in {"shutdown", "rundll32.exe"}:
            raise RuntimeError(f"Capture harness blocked unexpected process: {cmd}")
        self.calls.append(cmd)
        return std_subprocess.CompletedProcess(cmd, 0, stdout=b"", stderr=b"")

def settle(app: QApplication, delay: float = 0.08) -> None:
    deadline = time.time() + delay
    while time.time() < deadline:
        app.processEvents()
        time.sleep(0.01)
    app.processEvents()

def save_state(window, path: Path, app: QApplication) -> None:
    settle(app)
    pixmap = window.grab()
    if not pixmap.save(str(path), "PNG"):
        raise RuntimeError(f"Could not save capture: {path}")
~~~

The same file must implement capture_all() using shutdown_timer.subprocess = FakeSubprocessRun(), local CONFIG_FILE/WINDOW_CONFIG_FILE paths, QMessageBox.question returning QMessageBox.Yes, and native QT_QPA_PLATFORM=windows.
